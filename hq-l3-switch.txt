! ============================================================
! HQ-SW-L3 Configuration
! Enterprise Network Lab - HQ Layer 3 Switch
! ============================================================
! Device: Cisco Layer 3 Switch (IOSv-L2 with IP routing)
! Purpose: VLANs, SVIs, Inter-VLAN Routing, DHCP, Trunking
! ============================================================

enable
configure terminal
hostname HQ-SW-L3
no ip domain-lookup

! --- Enable Layer 3 Routing ---
ip routing

! --- VTY Access ---
line vty 0 15
 login local
 transport input ssh

! --- Local User ---
username admin privilege 15 secret cisco123

! ============================================================
! VLAN DATABASE
! ============================================================
vlan 10
 name IT_Department
vlan 20
 name HR_Department
vlan 30
 name Guest_Network
vlan 99
 name Management_Native

! ============================================================
! SVIs (Switch Virtual Interfaces) - Gateway for each VLAN
! ============================================================
interface Vlan10
 description IT_Gateway
 ip address 10.1.10.1 255.255.255.0
 ip helper-address 10.1.10.10
 no shutdown

interface Vlan20
 description HR_Gateway
 ip address 10.1.20.1 255.255.255.0
 ip helper-address 10.1.10.10
 no shutdown

interface Vlan30
 description Guest_Gateway
 ip address 10.1.30.1 255.255.255.0
 ip helper-address 10.1.10.10
 no shutdown

interface Vlan99
 description Management
 ip address 10.1.99.1 255.255.255.0
 no shutdown

! ============================================================
! UPLINK TO HQ ROUTER (Layer 3 Link)
! ============================================================
interface GigabitEthernet0/0
 description Uplink_to_HQ-RTR
 no switchport
 ip address 10.1.1.2 255.255.255.252
 no shutdown

! ============================================================
! TRUNK TO ACCESS SWITCH
! ============================================================
interface GigabitEthernet0/1
 description Trunk_to_HQ-SW-ACC
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,20,30,99
 no shutdown

! ============================================================
! OSPF ON L3 SWITCH
! ============================================================
router ospf 1
 router-id 1.1.1.2
 network 10.1.1.0 0.0.0.3 area 0
 network 10.1.10.0 0.0.0.255 area 0
 network 10.1.20.0 0.0.0.255 area 0
 network 10.1.30.0 0.0.0.255 area 0
 network 10.1.99.0 0.0.0.255 area 0
 passive-interface Vlan10
 passive-interface Vlan20
 passive-interface Vlan30
 passive-interface Vlan99

! ============================================================
! ACLs - SECURITY POLICIES
! ============================================================

! Guest VLAN cannot access IT or HR
ip access-list extended GUEST-ACL
 deny ip 10.1.30.0 0.0.0.255 10.1.10.0 0.0.0.255
 deny ip 10.1.30.0 0.0.0.255 10.1.20.0 0.0.0.255
 permit ip 10.1.30.0 0.0.0.255 any

! Apply Guest ACL to Guest SVI
interface Vlan30
 ip access-group GUEST-ACL in

! ============================================================
! SAVE
! ============================================================
end
write memory

! ============================================================
! VERIFICATION COMMANDS:
! ============================================================
! show vlan brief
! show ip interface brief
! show ip route
! show ip ospf neighbor
! show ip ospf database
! show access-lists
